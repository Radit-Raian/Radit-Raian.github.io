/** * COSMOS ENGINE CORE v3.0 * Features: * - Stellar temperature color mapping (H-R Diagram compliant) * - Multi-layer parallax scroll system * - Vector-based shooting star trajectories * - Dynamic Twinkle (Luminosity fluctuation) */ class CosmosEngine { constructor() { this.canvas = document.getElementById('cosmos-engine'); this.ctx = this.canvas.getContext('2d'); this.w = 0; this.h = 0; this.scrollY = 0; // Spectral types: O (Blue), B (Blue-White), A (White), F (Yellow-White), G (Yellow), K (Orange), M (Red) this.colors = ['#9bb0ff', '#aabfff', '#ffffff', '#f8f7ff', '#fff4ea', '#ffd2a1', '#ffcc6f']; this.stars = []; this.meteors = []; this.init(); } init() { this.resize(); window.addEventListener('resize', () => this.resize()); window.addEventListener('scroll', () => { this.scrollY = window.pageYOffset; }); // Layer 1: Background (Distant) this.generateStars(400, 0.03, 0.6); // Layer 2: Midground this.generateStars(250, 0.08, 1.1); // Layer 3: Foreground (Bright) this.generateStars(100, 0.18, 1.6); this.animate(); this.startMeteors(); } resize() { this.w = this.canvas.width = window.innerWidth; this.h = this.canvas.height = window.innerHeight; } generateStars(count, speed, maxRadius) { for (let i = 0; i < count; i++) { this.stars.push({ x: Math.random() * this.w, y: Math.random() * this.h, r: Math.random() * maxRadius + 0.1, color: this.colors[Math.floor(Math.random() * this.colors.length)], speed: speed, opacity: Math.random(), twinkle: (Math.random() * 0.02) + 0.005 }); } } startMeteors() { setInterval(() => { if (document.visibilityState === 'visible' && this.meteors.length < 3) { this.spawnMeteor(); } }, 4000); } spawnMeteor() { this.meteors.push({ x: Math.random() * this.w, y: Math.random() * (this.h * 0.4), vx: -(Math.random() * 8 + 5), vy: Math.random() * 8 + 5, len: Math.random() * 120 + 60, opacity: 1 }); } draw() { this.ctx.fillStyle = '#010409'; // Pure Dark this.ctx.fillRect(0, 0, this.w, this.h); // Update & Draw Stars this.stars.forEach(s => { s.opacity += s.twinkle; if (s.opacity > 1 || s.opacity < 0.2) s.twinkle *= -1; let py = (s.y + (this.scrollY * s.speed)) % this.h; if (py < 0) py += this.h; this.ctx.beginPath(); this.ctx.arc(s.x, py, s.r, 0, Math.PI * 2); this.ctx.fillStyle = s.color; this.ctx.globalAlpha = s.opacity; this.ctx.fill(); }); // Update & Draw Meteors this.ctx.globalAlpha = 1.0; for (let i = this.meteors.length - 1; i >= 0; i--) { let m = this.meteors[i]; this.ctx.strokeStyle = `rgba(255, 255, 255, ${m.opacity})`; this.ctx.lineWidth = 2; this.ctx.beginPath(); this.ctx.moveTo(m.x, m.y); this.ctx.lineTo(m.x + m.len, m.y - m.len); this.ctx.stroke(); m.x += m.vx; m.y += m.vy; m.opacity -= 0.015; if (m.opacity <= 0) this.meteors.splice(i, 1); } } animate() { this.draw(); requestAnimationFrame(() => this.animate()); } } new CosmosEngine();
